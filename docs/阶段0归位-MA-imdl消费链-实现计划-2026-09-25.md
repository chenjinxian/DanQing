```cpp
// Ported from: itwinjs-core PackedFeatureTable.ts constructor (:35-58).
PackedFeatureTable::PackedFeatureTable(std::vector<uint32_t> data, uint64_t modelId,
                                       uint32_t numFeatures, BatchType type)
    : m_data(std::move(data))
{
    m_numFeatures = numFeatures;
    m_modelId = modelId;
    m_type = type;
    // _subCategoriesOffset = 3 * numFeatures（PackedFeatureTable.ts:214 区）;
    // 尾部 2×u32/subcat。
    size_t const subCatsOffset = 3u * static_cast<size_t>(numFeatures);
    m_numSubCategories = m_data.size() > subCatsOffset
        ? static_cast<uint32_t>((m_data.size() - subCatsOffset) / 2u) : 0u;
}
```

```cpp
TileContent ImdlTile::readContent(uint8_t const* data, size_t dataSize)
{
    // Ported from: IModelTile.readContent (IModelTile.ts:94-131) + ImdlReader
    // (ImdlReader.ts:104-123 — decode → convertFeatureTable → createBatch).
    TileContent content;
    if (!data || dataSize < 4)
        return content;
    if (DetectTileFormat(data, dataSize) != TileFormat::IModel)
        return content;

    ImdlByteStream stream(data, dataSize);
    auto const header = ImdlHeader::readFrom(stream);
    if (!header.isValid())
        return content;

    // Feature table: read the 12B header + the packed words (3×u32/feature +
    // 2×u32/subcat tail = length-12 bytes) instead of skipping them.
    // Ported from: ParseImdlDocument.ts:1278-1284 (ftStartPos + FeatureTableHeader)
    // + FeatureTableHeader.ts layout (length includes the 12B header).
    ImdlFeatureTableHeader ftHeader;
    if (!ImdlFeatureTableHeader::readFrom(stream, ftHeader))
        return content;
    std::vector<uint32_t> featureWords;
    size_t const wordBytes = ftHeader.length >= ImdlFeatureTableHeader::sizeInBytes
        ? ftHeader.length - ImdlFeatureTableHeader::sizeInBytes : 0;
    featureWords.resize(wordBytes / 4);
    for (auto& w : featureWords) {
        w = stream.readUint32();   // 按 ImdlByteStream 既有读取 API 名调整
        if (stream.isPastTheEnd())
            return content;
    }

    auto const desc = decodeImdlContentDescriptionHeaderOnly(header);
    if (!desc.has_value())
        return content;
    content.contentRange = desc->contentRange;
    content.isLeaf = desc->isLeaf;

    auto doc = parseImdlDocument(stream, &ftHeader, &featureWords);
    if (!doc.has_value())
        return content;

    auto meshes = decodeImdlGraphics(*doc);
    if (meshes.empty())
        return content;

    RenderSystem* system = getTree().getRenderSystem();
    if (!system)
        return content;

    std::vector<RenderGraphic*> graphics;
    for (auto& polyface : meshes) {
        if (auto* graphic = system->createGraphicFromPolyface(polyface.Get(), 0xFFFFFFFFu, 0))
            graphics.push_back(graphic);
    }
    if (graphics.empty())
        return content;
    RenderGraphic* list = system->createGraphicList(std::move(graphics));

    // convertFeatureTable (ParseImdlDocument.ts:1259-1266)：线上 words →
    // PackedFeatureTable（DanQing 单模型子集：BatchType::Primary，modelId 0 —
    // 离线 tileset 无 model 语义；MultiModelPackedFeatureTable 登记 deferred，
    // ImdlFlags::MultiModelFeatureTable 置位时走既有空表路径并注释 TODO）。
    if (doc->featureCount > 0 && !doc->featureData.empty()) {
        dqCommon::PackedFeatureTable packed(doc->featureData, /*modelId*/0,
                                            doc->featureCount,
                                            dqCommon::BatchType::Primary);
        dqCommon::FeatureTable unpacked = packed.unpack();   // 值返回（拷贝省略）
        content.featureTable =
            std::make_unique<dqCommon::FeatureTable>(std::move(unpacked));
        // 若 FeatureTable 无移动构造（编译错误即为此情况），改用参考式逐条插入：
        //   auto ft = std::make_unique<dqCommon::FeatureTable>(
        //       packed.getNumFeatures(), dqBase::DqId{}, dqCommon::BatchType::Primary);
        //   for (uint32_t i = 0; i < packed.getNumFeatures(); ++i) {
        //       auto pf = packed.getFeature(i);          // PackedFeature
        //       dqCommon::Feature f{/*elementId*/dqBase::DqId(pf.elementId),
        //                           /*subCategoryId*/dqBase::DqId(pf.subCategoryId),
        //                           pf.geometryClass};
        //       ft->insertWithIndex(f, static_cast<int>(i));
        //   }
        //   content.featureTable = std::move(ft);
        // （Feature 构造的字段名以 dqCommon/FeatureTable.h 的 struct Feature 为准）

        // ImdlReader.ts:122-123：graphic = system.createBatch(graphic, ft, range)。
        content.graphic.reset(system->createBatch(list, content.featureTable.get(),
                                                  content.contentRange));
    } else {
        content.graphic.reset(list);
    }
    return content;
}
```
