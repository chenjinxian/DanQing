```js
// collector.mjs — RPC dump 落盘服务（零依赖，node http）。
// POST /tile   body=imdl 原始字节，header X-Dump-Meta=base64(JSON{treeId,contentId,guid,iModelId,changesetId,byteLength})
// POST /props  body=IModelTileTreeProps JSON，header X-Dump-Meta=base64(JSON{treeId,iModelId})
// POST /flush  body=JSON{formatVersion, itwinjsCommit?} → 聚合写 manifest.json
// GET  /health → 200（注入侧自检用）
import http from "node:http"; import fs from "node:fs"; import path from "node:path"; import crypto from "node:crypto";
const outDir = path.resolve(process.argv[2] ?? "dumps/session"); fs.mkdirSync(path.join(outDir,"files"),{recursive:true});
let seq = 0; const trees = [], tiles = [], meta0 = {};
const jf = n => path.join(outDir, "files", `${n}`);
const srv = http.createServer((req,res)=>{ const chunks=[]; req.on("data",c=>chunks.push(c)); req.on("end",()=>{
  const body=Buffer.concat(chunks); const meta=req.headers["x-dump-meta"]?JSON.parse(Buffer.from(req.headers["x-dump-meta"],"base64").toString()):{};
  if(req.url==="/health"){res.end("ok");return;}
  const n=seq++; fs.writeFileSync(jf(n), body);
  if(req.url==="/tile"){ const sha=crypto.createHash("sha256").update(body).digest("hex");
    tiles.push({treeId:meta.treeId,contentId:meta.contentId,guid:meta.guid,iModelId:meta.iModelId,changesetId:meta.changesetId,byteLength:body.length,sha256:sha256,file:`files/${n}`}); }
  else if(req.url==="/props"){ trees.push({treeId:meta.treeId,iModelId:meta.iModelId,file:`files/${n}`}); }
  else if(req.url==="/flush"){ Object.assign(meta0, meta||{}); }
  res.end("ok");
  // 周期/退出时聚合：process.on("SIGINT") → 写 manifest（provenance=meta0+trees+tiles），下条实现
});});
process.on("SIGINT",()=>{ fs.writeFileSync(path.join(outDir,"manifest.json"), JSON.stringify({provenance:meta0,trees,tiles},null,2)); console.error(`manifest written: ${trees.length} trees, ${tiles.length} tiles`); process.exit(0); });
srv.listen(8787,()=>console.error("collector on 8787 →", outDir));
// 注意：SIGINT 聚合有丢尾风险——每 N=32 条即时重写 manifest（幂等），SIGINT 只是最后一致化。
```

（实现时完善：即时重写 manifest 的节流逻辑；collector 生命周期归 capture.mjs。）
