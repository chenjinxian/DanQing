```cpp
// TD-21: reference is garbage-collected; C++ owns the array.
FeatureTable::~FeatureTable() { delete[] m_array; }

FeatureTable::FeatureTable(FeatureTable const& rhs)
    : m_modelId(rhs.m_modelId), m_type(rhs.m_type),
      m_maxFeatures(rhs.m_maxFeatures), m_size(rhs.m_size),
      m_array(new IndexedFeature[static_cast<size_t>(rhs.m_maxFeatures)])
{
    std::memcpy(m_array, rhs.m_array,
                sizeof(IndexedFeature) * static_cast<size_t>(rhs.m_size));
}

FeatureTable& FeatureTable::operator=(FeatureTable const& rhs)
{
    if (this != &rhs) {
        FeatureTable tmp(rhs);   // copy-and-swap: self-assign safe
        swap(tmp);
    }
    return *this;
}

FeatureTable::FeatureTable(FeatureTable&& rhs) noexcept
    : m_modelId(rhs.m_modelId), m_type(rhs.m_type),
      m_maxFeatures(rhs.m_maxFeatures), m_size(rhs.m_size), m_array(rhs.m_array)
{
    rhs.m_array = nullptr; rhs.m_size = 0; rhs.m_maxFeatures = 0;
}

FeatureTable& FeatureTable::operator=(FeatureTable&& rhs) noexcept
{
    if (this != &rhs) {
        delete[] m_array;
        m_modelId = rhs.m_modelId; m_type = rhs.m_type;
        m_maxFeatures = rhs.m_maxFeatures; m_size = rhs.m_size;
        m_array = rhs.m_array;
        rhs.m_array = nullptr; rhs.m_size = 0; rhs.m_maxFeatures = 0;
    }
    return *this;
}
```

`swap` 私有辅助（或用 std::swap 逐成员）：`void swap(FeatureTable& o) noexcept { std::swap(m_array,o.m_array); std::swap(m_size,o.m_size); std::swap(m_maxFeatures,o.m_maxFeatures); }`——DqId/BatchType 成员按现有类型逐个交换。

**核查既有消费面**：`unpack()` 值返回（prvalue 直接构造，合法）；M-A 的 `make_unique<FeatureTable>(std::move(unpacked))`（move 构造，合法）；`OpenGLRenderSystem::createBatch` 内部若经逐条 insertWithIndex 拷贝则不受影响，若用隐式拷贝构造则现在变深拷贝（正确化）。grep `FeatureTable` 全部拷贝点确认无浅别名依赖。

// Ported from: IModelTile.selectTiles (IModelTile.ts:205-334) — SelectParent
// 协议：跳级计数（maxInitialTilesToSkip/maxTilesToSkip）、NotFound 回退、
// 父子独占回滚、undisplayable root 特例。
// EQUIVALENCE: 参考源=IModelTile.ts:276 loadChildren 异步（返回 Promise +
//   TileTreeLoadStatus.Loading 态）；发散=DanQing loadChildren 同步执行
//   （无异步源，Loading 态不可达）→ markChildrenLoading 分支保留但当前
//   不可达；验证法=SelectTilesProtocol 场景矩阵 + TileTreeRender 像素锁。
//   另：参考 :211-213 的 debugMaxDepth 门（iModelTree.debugMaxDepth）DanQing
//   无 debug 面——省略登记。
SelectParent ImdlTile::selectTiles(std::vector<Tile*>& selected, TileDrawArgs& args,
                                   uint32_t numSkipped)
{
    TileVisibility vis = getTree().computeVisibility(args, this);
    if (vis == TileVisibility::OutsideFrustum)
        return SelectParent::No;

    if (vis == TileVisibility::Visible) {
        if (!isReady())
            args.insertMissing(this);

        if (hasGraphics()) {
            args.markReady(this);
            selected.push_back(this);
        } else if (!isReady()) {
            size_t const initialSize = selected.size();
            if (!hasLoadedChildren())
                return SelectParent::Yes;

            // 参考 :227-237 的循环形态逐字保留（含"首孩子后无条件 return
            // No"的参考原形——不"修正"它，§0）：
            if (getDepth() < getTree().getMaxInitialTilesToSkip()) {
                for (auto* kid : getChildren()) {
                    if (kid->selectTiles(selected, args, numSkipped) == SelectParent::Yes) {
                        selected.resize(initialSize);
                        return SelectParent::Yes;
                    }
                    return SelectParent::No;
                }
            }

            // 所有可见直接孩子可画则画它们（:240-250）。
            for (auto* kid : getChildren()) {
                if (getTree().computeVisibility(args, kid) != TileVisibility::OutsideFrustum) {
                    if (!kid->hasGraphics()) {
                        selected.resize(initialSize);
                        return SelectParent::Yes;
                    }
                    selected.push_back(kid);
                }
            }
            args.markUsed(this);
        }
        return SelectParent::No;
    }

    // TooCoarse：可能跳过本瓦直取后代（:254-262）。
    bool canSkipThisTile = (m_hadGraphics && !hasGraphics())
                           || getDepth() < getTree().getMaxInitialTilesToSkip();
    if (canSkipThisTile) {
        numSkipped = 1;
    } else {
        canSkipThisTile = isReady() || isParentDisplayable()
                          || getDepth() < getTree().getMaxInitialTilesToSkip();
        if (canSkipThisTile && isDisplayable()) {
            bool const isNotReady = !isReady() && !hasGraphics() && !(m_sizeMultiplier > 0.0);
            if (isNotReady) {
                if (numSkipped >= getTree().getMaxTilesToSkip())
                    canSkipThisTile = false;
                else
                    numSkipped += 1;
            }
        }
    }

    loadChildren();   // 同步（EQUIVALENCE 见上）；childrenLoadStatus 不可达 Loading
    bool const haveChildren = canSkipThisTile && hasLoadedChildren();
    if (canSkipThisTile /* && Loading —— 不可达，保留结构位 */)
        args.markUsed(this);

    if (haveChildren) {
        bool const undisplayableRoot = isUndisplayableRootTile();
        args.markUsed(this);
        bool drawChildren = true;
        size_t const initialSize = selected.size();
        for (auto* child : getChildren()) {
            // NB 必须遍历完所有孩子（missing 请求在递归内发出），参考 :283-285。
            if (child->selectTiles(selected, args, numSkipped) == SelectParent::Yes) {
                if (child->getLoadStatus() == TileLoadStatus::NotFound)
                    drawChildren = canSkipThisTile = false;
                else
                    drawChildren = undisplayableRoot;
            }
        }
        if (drawChildren)
            return SelectParent::No;
        if (args.parentsAndChildrenExclusive)
            selected.resize(initialSize);
    }

    if (isReady()) {
        if (hasGraphics()) {
            selected.push_back(this);
            if (!canSkipThisTile)
                args.markReady(this);
        }
        return SelectParent::No;
    }

    if (!canSkipThisTile)
        args.insertMissing(this);
    return isParentDisplayable() ? SelectParent::Yes : SelectParent::No;
}
